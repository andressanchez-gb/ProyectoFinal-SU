CREATE OR REPLACE FUNCTION fn_trg_validar_anio_vehiculo()
RETURNS TRIGGER AS $$
BEGIN
    -- # Validación: No permitir años de modelo mayores al año actual + 1
    IF NEW.anio > (EXTRACT(YEAR FROM CURRENT_DATE) + 1) THEN
        RAISE EXCEPTION 'Año del vehículo inválido. Excede el modelo del próximo año.';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_validar_anio_vh
BEFORE INSERT OR UPDATE ON tb_items
FOR EACH ROW
EXECUTE FUNCTION fn_trg_validar_anio_vehiculo();
